class Solution {
public:
    int findLucky(vector<int>& arr) {
        int luckynum = -1;

        for (int i = 0; i < arr.size(); i++) {
            int frequency = 0;

            for (int j = 0; j < arr.size(); j++) {
                if (arr[i] == arr[j]) {
                    frequency++; // count all occurrences
                }
            }

            // Check if number is lucky and bigger than previous lucky
            if (frequency == arr[i] && arr[i] > luckynum) {
                luckynum = arr[i];
            }
        }

        return luckynum;
    }
};
