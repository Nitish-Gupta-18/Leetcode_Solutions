class Solution {
public:
    bool lemonadeChange(vector<int>& bills) {
        int five_dollars = 0, ten_dollars = 0;
        
        for (int i = 0; i < bills.size(); i++) {
            int bill = bills[i];

            if (bill == 5) {
                five_dollars++;
            } else if (bill == 10) {
                if (five_dollars > 0) {
                    five_dollars--;
                    ten_dollars++;
                } else {
                    return false;
                }
            } else if (bill == 20) {
                if (ten_dollars > 0 && five_dollars > 0) {
                    ten_dollars--;
                    five_dollars--;
                } else if (five_dollars >= 3) {
                    five_dollars -= 3;
                } else {
                    return false;
                }
            }
        }
        
        return true;
    }
};
