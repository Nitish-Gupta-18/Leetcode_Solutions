class Solution {
public:
    int islandPerimeter(vector<vector<int>>& grid) {
        int n=grid.size();
        int m=grid[0].size();
        int count =0;
        for(int i=0;i<grid.size();i++){
            for(int j=0;j<grid[0].size();j++){
                if(grid[i][j]==1){
                    //to check left 
                    if(j>0&&grid[i][j-1]==0 || j==0){
                        count ++;
                    }
                    //to check right 
                    if(j<m-1&&grid[i][j+1]==0|| j==m-1){
                        count ++;
                    }
                    //to check top 
                    if(i>0&&grid[i-1][j]==0||i==0){
                        count ++;

                    }
                    //to check bottom
                    if(i<n-1&&grid[i+1][j]==0||i==n-1){
                        count++;
                    }
                }
            }
        }
        return count;
        
    }
};
