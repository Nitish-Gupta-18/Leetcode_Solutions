class Solution {
public:
    string convertToBase7(int num) {
        if (num == 0) return "0";

        bool isNegative = num < 0;
        num = abs(num);

        string result = "";

        while (num > 0) {
            int remainder = num % 7;
            result += to_string(remainder);
            num /= 7;
        }

        if (isNegative) result += '-';

        // Reverse the result to get correct order
        reverse(result.begin(), result.end());

        return result;
    }
};
