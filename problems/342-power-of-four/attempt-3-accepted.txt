class Solution {
public:
    bool isPowerOfFour(int n) {
        if (n <= 0) {
            return false;
        }

        for (; n % 4 == 0; n = n / 4) {
            // keep dividing by 4
        }

        if (n == 1) {
            return true;
        } else {
            return false;
        }
    }
};
