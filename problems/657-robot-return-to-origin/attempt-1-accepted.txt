class Solution {
public:
    bool judgeCircle(string moves) {
        int x = 0, y = 0;
        for(int i = 0; i < moves.length(); i++) {
            char u = moves[i];
            if(u == 'L') x++;
            else if(u == 'R') x--;
            else if(u == 'U') y++;
            else if(u == 'D') y--;
        }
        return (x == 0 && y == 0);
    }
};
