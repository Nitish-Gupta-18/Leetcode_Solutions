class Solution {
public:
    int arrangeCoins(int n) {
        long long usedCoins = 0; 

        for (long long row = 1; row <= n; row++) { 
            usedCoins += row;

            if (usedCoins == n)
                return row;

            if (usedCoins > n)
                return row - 1;
        }

        return 0;
    }
};
