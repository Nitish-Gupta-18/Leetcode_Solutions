class Solution {
public:
    int countKDifference(vector<int>& nums, int k) {
        int count=0;
        for(int i=0;i<nums.size(); i++){
            for(int j=i+1;j<nums.size();j++){
                if( abs(nums[i]-nums[j])==k ){ //The abs() function returns the absolute value of a number — that is, it removes the minus sign if the number is negative.
                    count ++;
                }
            }

        }
        return count;
        
    }
};
